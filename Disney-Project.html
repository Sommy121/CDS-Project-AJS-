ii.)

disney <- disney %>%
  mutate(period = cut(
    release_year,
    breaks = c(1900, 1925, 1950, 1975, 2000, 2025),
    labels = c("1900–1924", "1925–1949", "1950–1974", "1975–1999", "2000–2024"),
    right = FALSE
  ))
top_ratings <- disney %>%
  count(period, rating, sort = TRUE) %>%
  group_by(period) %>%
  slice_max(n, n = 5) %>%
  ungroup()
ratings_top <- disney %>%
  semi_join(top_ratings, by = c("period", "rating"))
ratings_top %>%
  count(period, release_year, rating) %>%
  ggplot(aes(x = release_year, y = n, fill = rating)) +
  geom_col(position = "fill") +
  facet_wrap(~ period, scales = "free_x") +
  labs(
    title = "Rating Distribution Over Time (Top 5 Ratings per 25-Year Interval)",
    x = "Year of Release",
    y = "Proportion",
    fill = "Rating"
  )

disney_genre <- disney %>%
  separate_rows(listed_in, sep = ", ") %>%
  mutate(listed_in = as.factor(listed_in))
disney_genre <- disney_genre %>%
  mutate(period = cut(
    release_year,
    breaks = c(1900, 1925, 1950, 1975, 2000, 2025),
    labels = c("1900–1924", "1925–1949", "1950–1974", "1975–1999", "2000–2024"),
    right = FALSE
  ))
top_genres <- disney_genre %>%
  count(period, listed_in, sort = TRUE) %>%
  group_by(period) %>%
  slice_max(n, n = 5) %>%
  ungroup()
disney_top <- disney_genre %>%
  semi_join(top_genres, by = c("period", "listed_in"))
disney_top %>%
  count(period, release_year, listed_in) %>%
  ggplot(aes(x = release_year, y = n, fill = listed_in)) +
  geom_col() +
  facet_wrap(~ period, scales = "free_x") +
  labs(
    title = "Top 5 Genres per 25-Year Interval",
    x = "Year of Release",
    y = "Number of Titles",
    fill = "Genre"
  )
